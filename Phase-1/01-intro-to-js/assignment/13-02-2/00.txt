Key Concepts to Remember
Object Methods:
const obj = {
    property: 'value',
    method: function() {
        return this.property; // 'this' refers to obj
    }
};
Classes:
class MyClass {
    #privateField = 0; // Private (ES2022)
    
    constructor(param) {
        this.publicField = param;
    }
    
    method() {
        return this.publicField;
    }
}
Inheritance:
class Child extends Parent {
    constructor(param1, param2) {
        super(param1); // Call parent constructor
        this.childProperty = param2;
    }
    
    method() {
        super.method(); // Call parent method
        // Additional code
    }
}
Static Properties:
class MyClass {
    static staticProperty = 'value';
    
    static staticMethod() {
        return MyClass.staticProperty;
    }
}

console.log(MyClass.staticProperty); // Access on class
